        async function handleAuthAction() {
            if (window.currentUser && !window.currentUser.isAnonymous) {
                try {
                    await window.firebaseAuth.signOut();
                    showToast("লগআউট সম্পন্ন হয়েছে।");
                } catch(e) {
                    console.error(e);
                }
            } else {
                try {
                    const provider = new (await import("https://www.gstatic.com/firebasejs/11.6.1/firebase-auth.js")).GoogleAuthProvider();
                    await (await import("https://www.gstatic.com/firebasejs/11.6.1/firebase-auth.js")).signInWithPopup(window.firebaseAuth, provider);
                } catch(err) {
                    console.warn("Popup error / fallback:", err);
                    simulateGuestGmailLogin();
                }
            }
        }

        async function loadCloudUserData(userId) {
            try {
                const db = window.firebaseDb;
                const appId = window.firebaseAppId;
                const { collection, getDocs } = await import("https://www.gstatic.com/firebasejs/11.6.1/firebase-firestore.js");

                const prodRef = collection(db, 'artifacts', appId, 'users', userId, 'products');
                const prodSnap = await getDocs(prodRef);
                const cloudProds = [];
                prodSnap.forEach(d => cloudProds.push({ id: d.id, ...d.data() }));
                productCatalog = cloudProds;

                const compRef = collection(db, 'artifacts', appId, 'users', userId, 'companies');
                const compSnap = await getDocs(compRef);
                const cloudComps = [];
                compSnap.forEach(d => cloudComps.push({ id: d.id, ...d.data() }));
                companies = cloudComps;

                const orderRef = collection(db, 'artifacts', appId, 'users', userId, 'orders');
                const orderSnap = await getDocs(orderRef);
                const cloudOrders = [];
                orderSnap.forEach(d => cloudOrders.push({ id: d.id, ...d.data() }));
                orderHistory = cloudOrders;

                renderProductCatalogUI();
                renderCompanies();
                renderOrderHistory();
            } catch (e) {
                console.warn("Cloud load error, using local storage:", e);
                loadLocalUserData();
            }
        }

        function loadLocalUserData() {
            try {
                const p = localStorage.getItem('user_private_products');
                productCatalog = p ? JSON.parse(p) : [];
                const c = localStorage.getItem('user_private_companies');
                companies = c ? JSON.parse(c) : [];
                const h = localStorage.getItem('user_private_orders');
                orderHistory = h ? JSON.parse(h) : [];
            } catch(e) {}
            renderProductCatalogUI();
            renderCompanies();
            renderOrderHistory();
        }

        function renderProductCatalogUI() {
            const dropdown = document.getElementById('productSelectDropdown');
            const badgesContainer = document.getElementById('quickProductBadges');

            dropdown.innerHTML = '<option value="">-- কোনো প্রোডাক্ট সিলেক্ট করা নেই --</option>';
            badgesContainer.innerHTML = '';

            if (productCatalog.length === 0) {
                badgesContainer.innerHTML = `
                    <div class="w-full text-center py-4 text-xs text-slate-400 border border-dashed border-slate-200 rounded-xl bg-slate-50/50">
                        আপনার অ্যাকাউন্টে এখনো কোনো প্রোডাক্ট নেই। উপরে <strong>"+ নতুন প্রোডাক্ট যুক্ত করুন"</strong> বোতামে ক্লিক করে তৈরি করুন।
                    </div>
                `;
                return;
            }

            const sortedProds = [...productCatalog].sort((a, b) => (b.pinned ? 1 : 0) - (a.pinned ? 1 : 0));

            sortedProds.forEach(prod => {
                const opt = document.createElement('option');
                opt.value = prod.id;
                opt.textContent = `${prod.pinned ? '⭐ ' : ''}${prod.name} — ৳${prod.price}`;
                dropdown.appendChild(opt);

                const badgeWrap = document.createElement('div');
                badgeWrap.className = `inline-flex items-center gap-1 px-2.5 py-1.5 rounded-xl border text-xs font-bold transition shadow-sm select-none ${
                    prod.pinned ? 'pinned-glow border-amber-300 text-amber-900' : 'bg-slate-50 hover:bg-slate-100 border-slate-200 text-slate-700'
                }`;

                badgeWrap.innerHTML = `
                    <button type="button" class="flex items-center gap-1.5 outline-none text-left" onclick="selectProductFromCatalog('${prod.id}')">
                        ${prod.pinned ? '<i class="fa-solid fa-star text-amber-500 text-[11px]"></i>' : '<i class="fa-solid fa-box text-slate-400 text-[10px]"></i>'}
                        <span class="truncate max-w-[120px]">${prod.name}</span>
                        <span class="bg-emerald-100 text-emerald-800 px-1.5 py-0.5 rounded text-[10px] font-extrabold">৳${prod.price}</span>
                    </button>
                    <button type="button" onclick="togglePinProduct(event, '${prod.id}')" class="text-slate-400 hover:text-amber-500 pl-1 text-[11px]" title="${prod.pinned ? 'আনপিন করুন' : 'পিন করুন'}">
                        <i class="${prod.pinned ? 'fa-solid text-amber-500' : 'fa-regular'} fa-bookmark"></i>
                    </button>
                    <button type="button" onclick="deleteProduct(event, '${prod.id}')" class="text-slate-300 hover:text-red-500 pl-1 text-[11px]" title="মুছে ফেলুন">
                        ✕
                    </button>
                `;
                badgesContainer.appendChild(badgeWrap);
            });
        }

        function selectProductFromCatalog(prodId) {
            const prod = productCatalog.find(p => p.id === prodId);
            if (!prod) return;

            document.getElementById('productSelectDropdown').value = prod.id;
            document.getElementById('productNameInput').value = prod.name;
            document.getElementById('productPriceInput').value = prod.price;

            state.productName = prod.name;
            state.productPrice = prod.price;

            handlePaymentChange();
            showToast(`সিলেক্ট হয়েছে: ${prod.name} (৳${prod.price})`);
        }

        function handleProductDropdownChange(prodId) {
            if (!prodId) {
                state.productName = '';
                state.productPrice = '';
                document.getElementById('productNameInput').value = '';
                document.getElementById('productPriceInput').value = '';
                handlePaymentChange();
                return;
            }
            selectProductFromCatalog(prodId);
        }

        async function togglePinProduct(e, prodId) {
            e.stopPropagation();
            const prod = productCatalog.find(p => p.id === prodId);
            if (!prod) return;
            prod.pinned = !prod.pinned;

            await persistProduct(prod);
            renderProductCatalogUI();
            showToast(prod.pinned ? "প্রোডাক্ট শীর্ষে পিন করা হয়েছে ⭐" : "আনপিন করা হয়েছে");
        }

        async function deleteProduct(e, prodId) {
            e.stopPropagation();
            productCatalog = productCatalog.filter(p => p.id !== prodId);
            
            if (window.currentUser && !window.currentUser.isAnonymous) {
                try {
                    const { doc, deleteDoc } = await import("https://www.gstatic.com/firebasejs/11.6.1/firebase-firestore.js");
                    await deleteDoc(doc(window.firebaseDb, 'artifacts', window.firebaseAppId, 'users', window.currentUser.uid, 'products', prodId));
                } catch(e){}
            } else {
                localStorage.setItem('user_private_products', JSON.stringify(productCatalog));
            }

            renderProductCatalogUI();
            showToast("প্রোডাক্ট তালিকা থেকে সরানো হয়েছে");
        }

        async function submitNewCustomProduct() {
            const name = document.getElementById('modalNewProductName').value.trim();
            const price = parseFloat(document.getElementById('modalNewProductPrice').value);
            const isPinned = document.getElementById('modalProductPin').checked;

            if (!name) {
                showToast("প্রোডাক্টের নাম লিখুন!");
                return;
            }
            if (isNaN(price) || price <= 0) {
                showToast("সঠিক প্রোডাক্টের দাম লিখুন!");
                return;
            }

            const newProd = {
                id: 'prod_' + Date.now(),
                name: name,
                price: price,
                pinned: isPinned,
                createdAt: Date.now()
            };

            productCatalog.unshift(newProd);
            await persistProduct(newProd);

            renderProductCatalogUI();
            closeAddProductModal();
            selectProductFromCatalog(newProd.id);
            showToast("আপনার প্রোডাক্ট সফলভাবে সেভ হয়েছে! 💾");
        }

        function updateDisplay() {
            document.getElementById('badgeDiv').textContent = state.division || '--';
            document.getElementById('badgeDist').textContent = state.district || '--';
            document.getElementById('badgeUpz').textContent = state.upazila || '--';
            document.getElementById('badgeUni').textContent = state.union || '--';
            document.getElementById('badgeWrd').textContent = state.ward || '--';
            document.getElementById('badgePlace').textContent = state.deliveryPlace || '--';

            const addressParts = [];
            if (state.deliveryPlace) addressParts.push(`ডেলিভারি স্থান: ${state.deliveryPlace}`);
            if (state.ward) addressParts.push(state.ward);
            if (state.union) addressParts.push(state.union);
            if (state.upazila) addressParts.push(`উপজেলা: ${state.upazila}`);
            if (state.district) addressParts.push(`জেলা: ${state.district}`);
            if (state.division) addressParts.push(`বিভাগ: ${state.division}`);

            let slipLines = [];

            if (state.selectedCompany) {
                slipLines.push(`[${state.selectedCompany.name}]`);
            }
            if (state.customerName) slipLines.push(`গ্রাহকের নাম: ${state.customerName}`);
            if (state.customerPhone) slipLines.push(`মোবাইল নম্বর: ${state.customerPhone}`);
            if (addressParts.length > 0) slipLines.push(`ঠিকানা: ${addressParts.join(', ')}`);

            if (state.productName) {
                slipLines.push(`প্রোডাক্ট: ${state.productName}`);
            }
            if (state.productPrice) {
                const priceNum = parseFloat(state.productPrice) || 0;
                const advNum = parseFloat(state.advancePayment) || 0;
                slipLines.push(`মোট মূল্য: ৳${priceNum}`);
                if (advNum > 0) {
                    slipLines.push(`অ্যাডভান্স পরিশোধ: ৳${advNum}`);
                    slipLines.push(`ক্যাশ অন ডেলিভারি (বকেয়া): ৳${state.dueCOD}`);
                } else {
                    slipLines.push(`ক্যাশ অন ডেলিভারি (COD): ৳${priceNum}`);
                }

            const previewEl = document.getElementById('fullSlipPreview');
            const statusTag = document.getElementById('statusTag');

            if (slipLines.length > 0) {
                previewEl.innerText = slipLines.join('\n');
                if (state.district && state.upazila && (state.ward || state.deliveryPlace)) {
                    statusTag.textContent = "অর্ডার প্রস্তুত ✓";
                    statusTag.className = "text-[10px] bg-emerald-100 text-emerald-800 font-bold px-2 py-0.5 rounded-full border border-emerald-300";
                } else {

                    statusTag.className = "text-[10px] bg-amber-100 text-amber-800 font-bold px-2 py-0.5 rounded-full border border-amber-300";
                }
            } else {
                previewEl.innerText = "নিচের ১ নম্বর ধাপ থেকে বিভাগ সিলেক্ট করুন...";
                statusTag.textContent = "বাছাই করুন";
                statusTag.className = "text-[10px] bg-slate-100 text-slate-600 font-semibold px-2 py-0.5 rounded-full";
            }
        }

                const div = document.createElement('div');
                div.className = `inline-flex items-center gap-1.5 px-2.5 py-1.5 rounded-xl text-xs font-bold border transition cursor-pointer select-none ${
                    isSelected ? 'company-active' : 'bg-slate-50 text-slate-700 hover:bg-slate-100 border-slate-200'
                } ${comp.pinned && !isSelected ? 'pinned-glow' : ''}`;

                div.innerHTML = `
                    <button type="button" class="flex items-center gap-1 outline-none" onclick="toggleSelectCompany('${comp.id}')">
                        ${comp.pinned ? '<i class="fa-solid fa-star text-amber-500 text-[10px]"></i>' : '<i class="fa-regular fa-building text-[10px] text-slate-400"></i>'}
                        <span>${comp.name}</span>
                    </button>
                    <button type="button" onclick="togglePinCompany(event, '${comp.id}')" class="text-slate-400 hover:text-amber-500 pl-1 text-[11px]" title="${comp.pinned ? 'আনপিন করুন' : 'পিন করুন'}">
                        <i class="${comp.pinned ? 'fa-solid text-amber-500' : 'fa-regular'} fa-bookmark"></i>
                    </button>
                    <button type="button" onclick="deleteCompany(event, '${comp.id}')" class="text-slate-300 hover:text-red-500 pl-1 text-[11px]" title="মুছে ফেলুন">
                        ✕
                    </button>
                `;
                list.appendChild(div);
            });
        }

        function toggleSelectCompany(compId) {
            if (state.selectedCompany && state.selectedCompany.id === compId) {
                state.selectedCompany = null;
            } else {
                state.selectedCompany = companies.find(c => c.id === compId) || null;
            }
            renderCompanies();
            updateDisplay();
        }

        function togglePinCompany(e, compId) {
            e.stopPropagation();
            const comp = companies.find(c => c.id === compId);
            if (comp) {
                comp.pinned = !comp.pinned;
                saveSavedCompanies();
                renderCompanies();
                showToast(comp.pinned ? "কোম্পানি শীর্ষে পিন করা হয়েছে ⭐" : "আনপিন করা হয়েছে");
            }
        }

        function deleteCompany(e, compId) {
            e.stopPropagation();
            companies = companies.filter(c => c.id !== compId);
            if (state.selectedCompany && state.selectedCompany.id === compId) {
                state.selectedCompany = null;
            }
            saveSavedCompanies();
            renderCompanies();
            updateDisplay();
            showToast("কোম্পানি মুছে ফেলা হয়েছে");
        }

        function openCompanyModal() {
            document.getElementById('modalCompanyName').value = '';
            document.getElementById('modalCompanyPin').checked = true;
            document.getElementById('companyModal').classList.remove('hidden');
            document.getElementById('modalCompanyName').focus();
        }

        function closeCompanyModal() {
            document.getElementById('companyModal').classList.add('hidden');
        }

        function submitNewCompany() {
            const name = document.getElementById('modalCompanyName').value.trim();
            if (!name) {
                showToast("কোম্পানির নাম লিখুন");
                return;
            }
            const isPinned = document.getElementById('modalCompanyPin').checked;
            const newComp = { id: 'c_' + Date.now(), name, pinned: isPinned };
            companies.push(newComp);
            state.selectedCompany = newComp;
            saveSavedCompanies();
            closeCompanyModal();
            renderCompanies();
            updateDisplay();
            showToast("নতুন কোম্পানি যোগ হয়েছে!");
        }

        function saveSavedCompanies() {
            try { localStorage.setItem('app_companies_list', JSON.stringify(companies)); } catch (e) {}
        }
        function saveOrderHistory() {
            try { localStorage.setItem('app_orders_backup', JSON.stringify(orderHistory)); } catch (e) {}
        }
